# Python — voice (any length)
from tts_module import synthesize_mp3, voice_for
open("hello.mp3", "wb").write(synthesize_mp3("مرحبا بكم", voice_for("ar-SA", "male")))

# Python — video, 4K
import video
project = video.validate_project({"ratio": "16:9", "resolution": "2160", "theme": "academy",
    "voice": {"lang": "ar-SA", "gender": "female"},
    "scenes": [{"layout": "title", "title": "درس جديد", "narration": "أهلاً بكم في هذا الدرس."}]})
video.render_project(project, "lesson.mp4", assets=None)
